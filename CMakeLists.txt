cmake_minimum_required(VERSION 3.24)
project(XbRecUnpackNative VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
    message(FATAL_ERROR "XbRecUnpack requires the Windows Cabinet API")
endif()
include(FetchContent)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib URL https://zlib.net/fossils/zlib-1.3.2.tar.gz
    URL_HASH SHA256=bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(zlib)
add_executable(XbRecUnpack XbRecUnpack/main.cpp XbRecUnpack/common.cpp XbRecUnpack/cabinet.cpp XbRecUnpack/remote.cpp XbRecUnpack/recovery.cpp XbRecUnpack/rom.cpp XbRecUnpack/images.cpp)
target_compile_features(XbRecUnpack PRIVATE cxx_std_20)
target_compile_definitions(XbRecUnpack PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_include_directories(XbRecUnpack PRIVATE ${zlib_SOURCE_DIR} ${zlib_BINARY_DIR})
target_link_libraries(XbRecUnpack PRIVATE Cabinet zlibstatic)
add_custom_command(TARGET XbRecUnpack POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    ${zlib_SOURCE_DIR}/LICENSE $<TARGET_FILE_DIR:XbRecUnpack>/LICENSE.zlib.txt)
if(MSVC)
    target_compile_options(XbRecUnpack PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_options(XbRecUnpack PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/XbRecUnpack/app.manifest")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(output_race_tests tests/output_race_tests.cpp)
    target_compile_features(output_race_tests PRIVATE cxx_std_20)
    target_compile_definitions(output_race_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(output_race_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    add_test(NAME output_race COMMAND output_race_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        foreach(suite IN ITEMS native recovery images)
            add_test(NAME ${suite} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_tests.py $<TARGET_FILE:XbRecUnpack>)
        endforeach()
    endif()
endif()
